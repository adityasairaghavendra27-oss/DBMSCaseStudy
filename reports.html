Complex aggregate queries (SUM, AVG, COUNT, GROUP BY) across Indian hubs

-- Executing Aggregation Query:
SELECT b.City, b.BranchName, COUNT(r.ReservationID) AS TotalBookings, SUM(p.Amount) AS TotalRevenue_INR FROM Branch b LEFT JOIN Reservation r ON b.BranchID = r.PickupBranchID LEFT JOIN Payment p ON r.ReservationID = p.ReservationID GROUP BY b.BranchID;
